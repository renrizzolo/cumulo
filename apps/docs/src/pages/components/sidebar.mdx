# Sidebar

A collapsible application sidebar primitive managing expanded and icon-only states, docked or floating variants, and width animations.

```tsx
import { Sidebar, SidebarToggle, useSidebar } from '@cumulo/core';
```

### Dual-tier navigation (rail and overlay or push)

A two-tier navigation pattern pairing a permanent slim rail with an adjacent secondary sidebar that toggles between `type="overlay"` (expanding on hover/focus over page content without layout shift) and `type="push"` (pinned in document flow, shifting content).

<ComponentPreview 
  frame
  title="Dual Sidebar (Rail & Overlay / Push)"
  code={`function DualSidebarDemo() {
  const [isPinned, setIsPinned] = useState(false);
  const [isHovered, setIsHovered] = useState(false);

return (

  <HStack width="full" align="stretch" gap="sm" style={{ minHeight: '260px' }}>
    {/* Container coordinating hover and focus across both tiers */}
    <div
      style={{ display: 'flex', height: '100%', position: 'relative' }}
      onPointerEnter={() => setIsHovered(true)}
      onPointerLeave={() => setIsHovered(false)}
      onFocus={() => setIsHovered(true)}
      onBlur={(e) => {
        if (!e.currentTarget.contains(e.relatedTarget)) {
          setIsHovered(false);
        }
      }}
    >
      {/* Tier 1: Permanent Slim Primary Rail */}
      <Sidebar
        variant="docked"
        defaultCollapsed
        hoverBehaviour="tooltip"
        width="48px"
        collapsedWidth="48px"
      >
        <Panel padding="xs" style={{ alignItems: 'center' }}>
          <SideNav aria-label="Primary rail">
            <SideNavItem icon={<DashboardIcon />} label="Dashboard" active />
            <SideNavItem icon={<IssuesIcon />} label="Issues" />
            <SideNavItem icon={<SettingsIcon />} label="Settings" />
          </SideNav>
        </Panel>
      </Sidebar>

    {/* Tier 2: Secondary Contextual Sidebar (Overlay or Push depending on Pin state) */}

      <Sidebar
        type={isPinned ? 'push' : 'overlay'}
        variant="docked"
        collapsed={isPinned ? false : !isHovered}
        width="200px"
        collapsedWidth="0px"
        style={{ left: '48px', height: 'auto' }}
      >
        <Panel padding="sm" divider="bottom">
          <SidebarHeader title={<Heading as="h4" size="sm">Issues</Heading>}>
            <Button
              size="xs"
              variant={isPinned ? 'primary' : 'ghost'}
              onClick={() => setIsPinned((prev) => !prev)}
              aria-label={isPinned ? 'Unpin sidebar' : 'Pin sidebar'}
            >
              <PinIcon pinned={isPinned} />
            </Button>
          </SidebarHeader>
        </Panel>
        <Panel scrollbar="thin" padding="sm">
          <SideNav aria-label="Secondary navigation">
            <SideNavItem label="All Issues" active />
            <SideNavItem label="Assigned to Me" />
            <SideNavItem label="Bookmarked" />
            <SideNavItem label="History" />
          </SideNav>
        </Panel>
      </Sidebar>
    </div>

    {/* Main Content Area */}

    <Panel flex={1} padding="md">
      <Heading as="h4" size="md">Dashboard Content</Heading>
      <Text size="sm">
        {isPinned
          ? 'The secondary sidebar is pinned into layout flow (push mode).'
          : 'Hover or focus rail items to expand the secondary overlay sidebar.'}
      </Text>
    </Panel>

  </HStack>

);
}`}

>

  <DualSidebarExample />
</ComponentPreview>
